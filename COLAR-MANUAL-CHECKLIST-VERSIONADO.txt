// ================================================================
// CHECKLIST VERSIONADO (Fat-Client, 06/08/2026) — dados fixos no
// bundle, sem round-trip por pergunta. Ver Checklist_Versionado.js
// (pwa/) para a origem/rastreabilidade completa deste bloco.

const CHECKLIST_VERSION_ATUAL = 'v1.0';
const POOL_PERGUNTAS = { /* EXATAMENTE o mesmo objeto da SEÇÃO 1 acima */ };
const VERSIONS = { /* EXATAMENTE o mesmo objeto da SEÇÃO 1 acima */ };
const NOMES_FASE = {
  fase1_seguranca_liberacao: 'Fase 1 de 3 — Segurança e Liberação',
  fase2_execucao_tecnica: 'Fase 2 de 3 — Execução Técnica',
  fase3_testes_entrega: 'Fase 3 de 3 — Testes e Entrega'
};

// Motor novo (fat-client, por fase). Some das perguntas ficam de fora
// se a OS não trouxer checklist_version reconhecida — nesse caso
// irParaChecklist() cai no fluxo antigo (getProximaPergunta ao vivo)
// em vez de travar a tela: republicar o backend (Checklist_Versionado.js)
// antes do frontend evita isso, mas se acontecer fora de ordem o
// checklist não para de funcionar pros técnicos.
const CHKV2 = {
  fases: [], faseAtualIndex: 0, osId: null, idSharePointOS: null, respostas: [], temNC: false,

  inicializar(os) {
    this.osId = os.id; this.idSharePointOS = os.idSharePoint || '';
    this.fases = []; this.faseAtualIndex = 0; this.respostas = []; this.temNC = false;
    const estruturaFases = VERSIONS[os.checklist_version];
    for (const [chaveFase, arrayIds] of Object.entries(estruturaFases)) {
      const perguntas = arrayIds.map(id => {
        if (!POOL_PERGUNTAS[id]) throw new Error('Pergunta ' + id + ' nao encontrada no POOL.');
        return Object.assign({ id: id }, POOL_PERGUNTAS[id]);
      });
      this.fases.push({ chave: chaveFase, perguntas: perguntas });
    }
    this.renderizarFase(0);
  },

  renderizarFase(index) {
    this.faseAtualIndex = index;
    const fase = this.fases[index];
    document.getElementById('chk-progresso').textContent = NOMES_FASE[fase.chave] || fase.chave;
    document.getElementById('chk-v2-fase-titulo').textContent = NOMES_FASE[fase.chave] || fase.chave;
    document.getElementById('chk-v2-btn-avancar').textContent =
      index === this.fases.length - 1 ? 'FINALIZAR CHECKLIST' : 'PRÓXIMA FASE';
    const cont = document.getElementById('chk-v2-perguntas');
    cont.innerHTML = fase.perguntas.map(p => {
      const opcoesHtml = p.opcoes.map((op, i) =>
        '<label style="display:flex;align-items:center;gap:12px;padding:14px;background:var(--bg2);border:1px solid var(--border);border-radius:var(--radius);margin-bottom:8px;cursor:pointer">' +
        '<input type="radio" name="chkv2_' + p.id + '" value="' + i + '" data-nc="' + (op.nc ? '1' : '0') + '" ' +
          'onchange="CHKV2.onResponder(\'' + p.id + '\')" style="width:20px;height:20px;accent-color:var(--green)">' +
        '<span style="font-size:15px;color:var(--text)">' + op.label + '</span></label>'
      ).join('');
      return '<div class="card" style="margin-bottom:14px" id="chk-v2-card-' + p.id + '">' +
        '<div style="font-size:16px;font-weight:600;color:var(--text);margin-bottom:10px">' + p.texto + '</div>' +
        opcoesHtml +
        '<div id="chk-v2-nc-' + p.id + '" style="display:none;margin-top:6px">' +
          '<div class="alert alert-warn" style="margin-bottom:8px">Nao conformidade detectada nesta resposta.</div>' +
          '<textarea class="form-textarea" id="chk-v2-just-' + p.id + '" placeholder="Justificativa (obrigatoria)"></textarea>' +
          '<div class="alert alert-warn" style="margin-top:8px">Foto do item tambem obrigatoria — captura ainda nao disponivel nesta versao do app. Enquanto isso, esta pergunta nao pode ser confirmada como NC.</div>' +
        '</div></div>';
    }).join('');
    document.getElementById('chk-v2-erro-versao').style.display = 'none';
    document.getElementById('chk-v2-block').style.display = 'block';
    document.getElementById('chk-pergunta-block').style.display = 'none';
    document.getElementById('chk-resumo-block').style.display = 'none';
    document.getElementById('chk-loading').style.display = 'none';
  },

  onResponder(perguntaId) {
    const marcado = document.querySelector('input[name="chkv2_' + perguntaId + '"]:checked');
    const nc = marcado && marcado.dataset.nc === '1';
    document.getElementById('chk-v2-nc-' + perguntaId).style.display = nc ? 'block' : 'none';
  },

  tentarAvancarFase() {
    const fase = this.fases[this.faseAtualIndex];
    for (const p of fase.perguntas) {
      const marcado = document.querySelector('input[name="chkv2_' + p.id + '"]:checked');
      if (!marcado) {
        toast('A pergunta "' + p.texto + '" e obrigatoria', true);
        document.getElementById('chk-v2-card-' + p.id).scrollIntoView({ behavior: 'smooth', block: 'center' });
        return;
      }
      const opcao = p.opcoes[Number(marcado.value)];
      if (opcao.nc) {
        // Foto nunca existe no app hoje — NC fica permanentemente bloqueada
        // ate a captura de foto ser construida (limitacao conhecida, mesmo
        // tratamento do KM por-OS).
        toast('NC em "' + p.texto + '" exige foto do item, que este app ainda nao captura — nao e possivel avancar', true);
        document.getElementById('chk-v2-card-' + p.id).scrollIntoView({ behavior: 'smooth', block: 'center' });
        return;
      }
      this.respostas.push({ perguntaId: p.id, texto: p.texto, resposta: opcao.label, nc: opcao.nc });
    }
    if (this.faseAtualIndex === this.fases.length - 1) {
      this.finalizar();
    } else {
      this.renderizarFase(this.faseAtualIndex + 1);
    }
  },

  finalizar() {
    showLoading('Salvando checklist...');
    const chamadas = this.respostas.map(r =>
      gsCall('salvarResposta', [this.osId, this.idSharePointOS, r.perguntaId, r.texto, r.resposta, '', APP.tecnico.nome, r.nc], { enfileiravelSeOffline: true })
    );
    Promise.all(chamadas).then(() => {
      hideLoading();
      CHK.totalRespostas = this.respostas.length;
      CHK.temNC = this.respostas.some(r => r.nc);
      document.getElementById('chk-v2-block').style.display = 'none';
      document.getElementById('chk-resumo-block').style.display = 'block';
      document.getElementById('chk-resumo-qtd').textContent = CHK.totalRespostas + ' pergunta(s) respondida(s)';
      document.getElementById('chk-nc-aviso').style.display = CHK.temNC ? 'block' : 'none';
      document.getElementById('chk-progresso').textContent = 'Concluido';
    }).catch(() => { hideLoading(); toast('Erro ao salvar checklist', true); });
  }
};

function irParaChecklist() {
  const os = APP.osAtiva;
  if (!os) { toast('Nenhuma OS ativa', true); return; }
  document.getElementById('chk-loading').style.display = 'block';
  document.getElementById('chk-pergunta-block').style.display = 'none';
  document.getElementById('chk-resumo-block').style.display = 'none';
  document.getElementById('chk-v2-block').style.display = 'none';
  document.getElementById('chk-v2-erro-versao').style.display = 'none';
  document.getElementById('chk-progresso').textContent = 'Carregando...';
  showScreen('scr-checklist');

  if (os.checklist_version && VERSIONS[os.checklist_version]) {
    try {
      CHKV2.inicializar(os);
      return;
    } catch (e) {
      document.getElementById('chk-loading').style.display = 'none';
      document.getElementById('chk-v2-erro-msg').textContent =
        'Erro ao montar o checklist versionado (' + e.message + '). Peça para o time de TI conferir Checklist_Versionado.js.';
      document.getElementById('chk-v2-erro-versao').style.display = 'block';
      return;
    }
  }
  // Sem checklist_version reconhecida: fluxo antigo, sem qualquer bloqueio de tela.
  CHK.osId = os.id; CHK.idSharePointOS = os.idSharePoint || '';
  CHK.disciplina = 'NR-10'; CHK.perguntaAtualId = null; CHK.totalRespostas = 0; CHK.temNC = false;
  carregarPergunta(null, null);
}
function voltarDaChecklist() { showScreen('scr-os-ativa'); }
